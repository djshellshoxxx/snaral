cmake_minimum_required(VERSION 3.22)
project(Snaral VERSION 0.2.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
endif()

# ---------------------------------------------------------------------------
# Unit tests: every tests/*Tests.cpp is its own executable linking Source/ headers only
# (JUCE-independent). Tests needing JUCE live in tests/juce/*Tests.cpp (see below).
# ---------------------------------------------------------------------------
include(CTest)
file(GLOB SNARAL_CORE_SOURCES CONFIGURE_DEPENDS Source/Core/*.cpp Source/Engine/*.cpp Source/Dsp/*.cpp)
file(GLOB SNARAL_TEST_SOURCES CONFIGURE_DEPENDS tests/*Tests.cpp)

# >>> BEGIN state worker: JUCE-free state library + embedded factory presets >>>
# Source/State/Pure/*.cpp are JUCE-free (preset JSON, undo, variations, A/B, library, browser model) and are
# built once as a static library that every later target links (link_libraries below), so JUCE-free tests
# can use them without recompiling. Factory presets (Resources/presets/*.json) are embedded as byte arrays
# by pure CMake at configure time into <build>/generated/FactoryPresetBlobs.cpp: no JUCE binary-data tool,
# works on every OS/generator and cross builds. Re-run tools/gen_presets.py after any registry change.
file(GLOB SNARAL_STATE_PURE_SOURCES CONFIGURE_DEPENDS Source/State/Pure/*.cpp)
file(GLOB SNARAL_FACTORY_PRESET_FILES CONFIGURE_DEPENDS Resources/presets/*.json)
list(SORT SNARAL_FACTORY_PRESET_FILES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SNARAL_FACTORY_PRESET_FILES})
set(SNARAL_FACTORY_BLOB_ARRAYS "")
set(SNARAL_FACTORY_BLOB_TABLE "")
set(SNARAL_FACTORY_BLOB_COUNT 0)
foreach(preset_file ${SNARAL_FACTORY_PRESET_FILES})
    get_filename_component(preset_name "${preset_file}" NAME)
    file(READ "${preset_file}" preset_hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," preset_bytes "${preset_hex}")
    string(APPEND SNARAL_FACTORY_BLOB_ARRAYS "static const unsigned char blob${SNARAL_FACTORY_BLOB_COUNT}[] = {${preset_bytes}0x00};\n")
    string(APPEND SNARAL_FACTORY_BLOB_TABLE "    { \"${preset_name}\", blob${SNARAL_FACTORY_BLOB_COUNT}, sizeof (blob${SNARAL_FACTORY_BLOB_COUNT}) - 1 },\n")
    math(EXPR SNARAL_FACTORY_BLOB_COUNT "${SNARAL_FACTORY_BLOB_COUNT} + 1")
endforeach()
set(SNARAL_FACTORY_BLOB_CPP "${CMAKE_BINARY_DIR}/generated/FactoryPresetBlobs.cpp")
file(CONFIGURE OUTPUT "${SNARAL_FACTORY_BLOB_CPP}" @ONLY CONTENT
"// GENERATED by CMake from Resources/presets/*.json. Do not edit.
#include \"State/Pure/FactoryPresets.h\"

namespace snaral::embedded
{
@SNARAL_FACTORY_BLOB_ARRAYS@
const PresetBlob factoryPresetBlobs[] = {
@SNARAL_FACTORY_BLOB_TABLE@    { nullptr, nullptr, 0 }
};
const int factoryPresetBlobCount = @SNARAL_FACTORY_BLOB_COUNT@;
} // namespace snaral::embedded
")
add_library(snaral_state_pure STATIC ${SNARAL_STATE_PURE_SOURCES} "${SNARAL_FACTORY_BLOB_CPP}")
target_include_directories(snaral_state_pure PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Source")
target_compile_definitions(snaral_state_pure PUBLIC "SNARAL_SOURCE_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}\"")
set_target_properties(snaral_state_pure PROPERTIES POSITION_INDEPENDENT_CODE ON)
link_libraries(snaral_state_pure)
# <<< END state worker <<<

# >>> BEGIN engine-integration: core engine/DSP sources are compiled ONCE into a static library that every JUCE-free
# test links (instead of recompiling SnaralEngine.cpp into each of the test executables) >>>
add_library(snaral_core STATIC ${SNARAL_CORE_SOURCES})
target_include_directories(snaral_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Source")
set_target_properties(snaral_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
# <<< END engine-integration <<<

foreach(test_src ${SNARAL_TEST_SOURCES})
    get_filename_component(test_name ${test_src} NAME_WE)
    string(TOLOWER ${test_name} test_target)
    add_executable(snaral_${test_target} ${test_src})
    target_include_directories(snaral_${test_target} PRIVATE Source)
    target_link_libraries(snaral_${test_target} PRIVATE snaral_core)
    add_test(NAME snaral_${test_target} COMMAND snaral_${test_target})
endforeach()

option(SNARAL_BUILD_PLUGIN "Build plugin and standalone targets" ON)
option(SNARAL_BUILD_CLAP "Also build a CLAP plugin" ON)
# ci-speed: build-speed options. LTO and unity are switchable so PR/branch CI and local dev builds can skip the slow parts.
option(SNARAL_ENABLE_LTO "Link-time optimisation for the plugin (juce_recommended_lto_flags); ON for release builds" ON)
option(SNARAL_UNITY_BUILD "Batch Source/*.cpp into unity files for the plugin target (not tests, not JUCE modules)" ON)
set(SNARAL_UNITY_BATCH_SIZE 8 CACHE STRING "Files per unity batch")
set(SNARAL_JUCE_DIR "" CACHE PATH "Use a local JUCE checkout instead of fetching")
set(SNARAL_CLAP_EXT_DIR "" CACHE PATH "Use a local clap-juce-extensions checkout instead of fetching")

if(SNARAL_BUILD_PLUGIN)
    include(FetchContent)
    if(SNARAL_JUCE_DIR)
        add_subdirectory(${SNARAL_JUCE_DIR} juce EXCLUDE_FROM_ALL)
    else()
        FetchContent_Declare(JUCE
            GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
            GIT_TAG 9.0.3
            GIT_SHALLOW TRUE)
        FetchContent_MakeAvailable(JUCE)
    endif()

    if(SNARAL_BUILD_CLAP)
        if(SNARAL_CLAP_EXT_DIR)
            add_subdirectory(${SNARAL_CLAP_EXT_DIR} clap-juce-extensions EXCLUDE_FROM_ALL)
        else()
            # ci-speed: pinned to a commit (was GIT_TAG main). tools/ci_fetch_deps.sh reads SNARAL_CLAP_EXT_COMMIT below.
            set(SNARAL_CLAP_EXT_COMMIT 7adee3a1bd4684d4caa5601100e364abccff4b4d)
            FetchContent_Declare(clap-juce-extensions
                GIT_REPOSITORY https://github.com/free-audio/clap-juce-extensions.git
                GIT_TAG ${SNARAL_CLAP_EXT_COMMIT}
                GIT_SHALLOW FALSE)
            FetchContent_MakeAvailable(clap-juce-extensions)
        endif()
    endif()

    if(APPLE)
        set(SNARAL_FORMATS VST3 AU Standalone)
    else()
        set(SNARAL_FORMATS VST3 Standalone)
    endif()
    if(UNIX AND NOT APPLE)
        list(APPEND SNARAL_FORMATS LV2)
    endif()

    juce_add_plugin(Snaral
        COMPANY_NAME "Circuit Drift Labs"
        BUNDLE_ID "com.circuitdriftlabs.snaral"
        PLUGIN_MANUFACTURER_CODE Cdlb
        PLUGIN_CODE Snrl
        FORMATS ${SNARAL_FORMATS}
        AU_MAIN_TYPE kAudioUnitType_MusicDevice
        IS_SYNTH TRUE
        NEEDS_MIDI_INPUT TRUE
        NEEDS_MIDI_OUTPUT FALSE
        IS_MIDI_EFFECT FALSE
        EDITOR_WANTS_KEYBOARD_FOCUS TRUE
        VST3_CATEGORIES Instrument Synth
        LV2URI "https://circuitdriftlabs.com/plugins/snaral"
        COPY_PLUGIN_AFTER_BUILD FALSE
        LV2URI "https://circuitdriftlabs.com/plugins/snaral"
        PRODUCT_NAME "Snaral")

    # Application sources: every .cpp under Source/ (including Source/Core and Source/Engine).
    file(GLOB_RECURSE SNARAL_PLUGIN_SOURCES CONFIGURE_DEPENDS Source/*.cpp)
    target_sources(Snaral PRIVATE ${SNARAL_PLUGIN_SOURCES})
    target_include_directories(Snaral PRIVATE Source)
    target_compile_definitions(Snaral PUBLIC
        JUCE_WEB_BROWSER=0
        JUCE_USE_CURL=0
        JUCE_VST3_CAN_REPLACE_VST2=0
        JUCE_MODAL_LOOPS_PERMITTED=1   # lets the editor tests pump the message loop (MessageManager::runDispatchLoopUntil)
        JUCE_DISPLAY_SPLASH_SCREEN=0
        JUCE_REPORT_APP_USAGE=0
        JUCE_STRICT_REFCOUNTEDPOINTER=1)
    target_link_libraries(Snaral PRIVATE
        juce::juce_audio_utils
        juce::juce_dsp
        PUBLIC juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags)
    if(SNARAL_ENABLE_LTO)  # ci-speed
        target_link_libraries(Snaral PUBLIC juce::juce_recommended_lto_flags)
    endif()
    if(SNARAL_UNITY_BUILD)  # ci-speed: Snaral shared-code target only (tests and JUCE modules untouched)
        set_target_properties(Snaral PROPERTIES UNITY_BUILD ON UNITY_BUILD_BATCH_SIZE ${SNARAL_UNITY_BATCH_SIZE})
    endif()
    juce_generate_juce_header(Snaral)

    if(SNARAL_BUILD_CLAP)
        clap_juce_extensions_plugin(TARGET Snaral
            CLAP_ID "com.circuitdriftlabs.snaral"
            CLAP_FEATURES instrument synthesizer bass stereo
            CLAP_MISC_FLAGS "")
    endif()

    # JUCE-dependent tests: tests/juce/*Tests.cpp link juce_core/juce_data_structures + Source/ (non-GUI).
    file(GLOB SNARAL_JUCE_TEST_SOURCES CONFIGURE_DEPENDS tests/juce/*Tests.cpp)
    file(GLOB_RECURSE SNARAL_JUCE_TEST_LIB_SOURCES CONFIGURE_DEPENDS Source/Core/*.cpp Source/State/*.cpp Source/Parameters/*.cpp)
    foreach(test_src ${SNARAL_JUCE_TEST_SOURCES})
        get_filename_component(test_name ${test_src} NAME_WE)
        string(TOLOWER ${test_name} test_target)
        add_executable(snaral_juce_${test_target} ${test_src} ${SNARAL_JUCE_TEST_LIB_SOURCES})
        target_include_directories(snaral_juce_${test_target} PRIVATE Source)
        target_compile_definitions(snaral_juce_${test_target} PRIVATE JUCE_STANDALONE_APPLICATION=1
            JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
        target_link_libraries(snaral_juce_${test_target} PRIVATE
            juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags)
        add_test(NAME snaral_juce_${test_target} COMMAND snaral_juce_${test_target})
    endforeach()

    # >>> BEGIN engine-integration: processor-level tests link the plugin's shared code (tests/juce_plugin/*Tests.cpp) >>>
    file(GLOB SNARAL_PLUGIN_TEST_SOURCES CONFIGURE_DEPENDS tests/juce_plugin/*Tests.cpp)
    foreach(test_src ${SNARAL_PLUGIN_TEST_SOURCES})
        get_filename_component(test_name ${test_src} NAME_WE)
        string(TOLOWER ${test_name} test_target)
        add_executable(snaral_plugin_${test_target} ${test_src})
        target_include_directories(snaral_plugin_${test_target} PRIVATE Source tests "${CMAKE_BINARY_DIR}/Snaral_artefacts/JuceLibraryCode")
        target_link_libraries(snaral_plugin_${test_target} PRIVATE Snaral)
        target_include_directories(snaral_plugin_${test_target} PRIVATE "${SNARAL_JUCE_DIR}/modules")
        target_compile_definitions(snaral_plugin_${test_target} PRIVATE
            $<TARGET_PROPERTY:Snaral,COMPILE_DEFINITIONS> $<TARGET_PROPERTY:Snaral,INTERFACE_COMPILE_DEFINITIONS>)
        add_test(NAME snaral_plugin_${test_target} COMMAND snaral_plugin_${test_target})
    endforeach()
    # <<< END engine-integration <<<
endif()
